## 🚀 **COMO USAR**

### **1. Marketplace**
```javascript
import { serviceCatalog } from './marketplace/ServiceCatalog';

// Buscar serviços
const services = serviceCatalog.searchServices('financeiro', {
  category: 'financial',
  minPrice: 1000
});

// Calcular preço dinâmico
const pricing = serviceCatalog.calculateDynamicPrice('financial_diagnosis', {
  quantity: 2,
  urgency: 'high'
});

// Criar pedido
const orderId = serviceCatalog.createOrder({
  clientId: 'client123',
  serviceId: 'financial_diagnosis',
  quantity: 1
});
```

### **2. API Pública**
```javascript
import { publicAPI } from './api/PublicAPI';

// Gerar chave de API
const apiKey = publicAPI.generateAPIKey({
  clientId: 'client123',
  clientName: 'Test Client',
  tier: 'premium'
});

// Processar requisição
const response = await publicAPI.processRequest(
  'GET',
  '/api/v1/services',
  { 'x-api-key': apiKey.key }
);
```

### **3. SDK**
```javascript
import { EvoctoSDK } from './sdk/EvoctoSDK';

const sdk = new EvoctoSDK({
  apiKey: 'evocto_your_api_key_here',
  baseURL: 'https://api.evocto.com'
});

// Listar serviços
const services = await sdk.getServices({ category: 'financial' });

// Criar pedido
const order = await sdk.createOrder({
  serviceId: 'financial_diagnosis',
  clientInfo: {
    name: 'João Silva',
    email: 'joao@empresa.com'
  }
});
```

### **4. PWA**
```javascript
import { pwaManager } from './pwa/PWAManager';

// Verificar status
const status = pwaManager.getPWAStatus();

// Solicitar permissão de notificação
const permission = await pwaManager.requestNotificationPermission();

// Enviar notificação
await pwaManager.sendNotification('Título', {
  body: 'Corpo da notificação',
  icon: '/icon-192x192.png'
});
```

### **5. Integrações**
```javascript
import { integrationSystem } from './integrations/IntegrationSystem';

// Conectar integração
const connectionId = await integrationSystem.connectIntegration('salesforce', {
  access_token: 'your_token',
  refresh_token: 'your_refresh_token'
});

// Sincronizar dados
const syncJob = await integrationSystem.syncData(connectionId, 'clients');

// Configurar webhook
const webhookId = await integrationSystem.configureWebhook(connectionId, {
  url: 'https://example.com/webhook',
  events: ['client_created']
});
```
